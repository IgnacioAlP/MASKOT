def obtener_asistencias():
    """Obtener historial de asistencias"""
    if 'usuario_id' not in session or session.get('rol') not in ['dueÃ±o', 'admin']:
        return jsonify({'error': 'No autorizado'}), 403
    
    try:
        fecha_desde = request.args.get('fecha_desde', date.today().strftime('%Y-%m-%d'))
        fecha_hasta = request.args.get('fecha_hasta', date.today().strftime('%Y-%m-%d'))
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        cursor.execute("""
            SELECT a.id, u.username, a.fecha, a.hora_entrada, a.hora_salida,
                   TIME_TO_SEC(TIMEDIFF(a.hora_salida, a.hora_entrada))/3600 as horas_trabajadas
            FROM asistencia a
            LEFT JOIN personal p ON a.personal_id = p.id
            LEFT JOIN usuarios u ON p.usuario_id = u.id
            WHERE DATE(a.fecha) BETWEEN %s AND %s
            ORDER BY a.fecha DESC, a.hora_entrada DESC
        """, (fecha_desde, fecha_hasta))
        
        asistencias = cursor.fetchall()
        cursor.close()
        conexion.close()
        
        asistencias_list = []
        for asistencia in asistencias:
            asistencias_list.append({
                'id': asistencia[0],
                'usuario': asistencia[1],
                'fecha': asistencia[2].strftime('%Y-%m-%d'),
                'hora_entrada': str(asistencia[3]) if asistencia[3] else None,
                'hora_salida': str(asistencia[4]) if asistencia[4] else None,
                'horas_trabajadas': round(asistencia[5], 2) if asistencia[5] else 0
            })
        
        return jsonify({'asistencias': asistencias_list})
    
    except Exception as e:
        return jsonify({'error': f'Error al obtener asistencias: {str(e)}'}), 500

def obtener_resumen_asistencia():
    """Obtener resumen de asistencia por empleado"""
    if 'usuario_id' not in session or session.get('rol') not in ['dueÃ±o', 'admin']:
        return jsonify({'error': 'No autorizado'}), 403
    
    try:
        mes = request.args.get('mes', date.today().strftime('%Y-%m'))
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        cursor.execute("""
            SELECT u.username,
                   COUNT(a.id) as dias_trabajados,
                   SUM(TIME_TO_SEC(TIMEDIFF(a.hora_salida, a.hora_entrada))/3600) as total_horas,
                   AVG(TIME_TO_SEC(TIMEDIFF(a.hora_salida, a.hora_entrada))/3600) as promedio_horas_dia
            FROM usuarios u
            JOIN personal p ON u.id = p.usuario_id
            LEFT JOIN asistencia a ON p.id = a.personal_id AND DATE_FORMAT(a.fecha, '%Y-%m') = %s
            WHERE p.activo = TRUE
            GROUP BY u.id, u.username
        """, (mes,))
        
        resumen = cursor.fetchall()
        cursor.close()
        conexion.close()
        
        resumen_list = []
        for emp in resumen:
            resumen_list.append({
                'usuario': emp[0],
                'dias_trabajados': emp[1] or 0,
                'total_horas': round(emp[2] or 0, 2),
                'promedio_horas_dia': round(emp[3] or 0, 2)
            })
        
        return jsonify({'resumen': resumen_list})
    
    except Exception as e:
        return jsonify({'error': f'Error al obtener resumen: {str(e)}'}), 500

def obtener_citas():
    conexion = obtener_conexion()
    citas = []
    with conexion.cursor() as cursor:
        cursor.execute("""
            SELECT c.id, c.cliente_nombre, c.cliente_email, 
                   GROUP_CONCAT(DISTINCT s.nombre SEPARATOR ', ') as servicios, 
                   c.fecha, c.hora, c.estado,
                   GROUP_CONCAT(DISTINCT m.nombre SEPARATOR ', ') as mascotas,
                   c.precio_total, c.observaciones
            FROM citas c
            LEFT JOIN citas_servicios cs ON c.id = cs.cita_id
            LEFT JOIN servicios s ON cs.servicio_id = s.id
            LEFT JOIN citas_mascotas cm ON c.id = cm.cita_id
            LEFT JOIN mascotas m ON cm.mascota_id = m.id
            GROUP BY c.id, c.cliente_nombre, c.cliente_email, c.fecha, c.hora, c.estado
            ORDER BY c.fecha, c.hora
        """)
        citas = cursor.fetchall()
    conexion.close()
    return citas

def insertar_cita_completa(cliente_nombre, cliente_email, mascotas_data, servicios_ids, fecha, hora, observaciones=''):
    """Inserta una cita con mÃºltiples mascotas y servicios"""
    conexion = obtener_conexion()
    cita_id = None
    try:
        with conexion.cursor() as cursor:
            # Verificar/insertar cliente
            cliente_id = verificar_o_insertar_cliente(cursor, cliente_nombre, cliente_email)
            
            # Insertar cita principal
            cursor.execute("""
                INSERT INTO citas (cliente_id, cliente_nombre, cliente_email, fecha, hora, estado, observaciones) 
                VALUES (%s, %s, %s, %s, %s, 'pendiente', %s)
            """, (cliente_id, cliente_nombre, cliente_email, fecha, hora, observaciones))
            cita_id = cursor.lastrowid
            
            # Insertar mascotas y vincular con la cita
            for mascota_data in mascotas_data:
                mascota_id = insertar_o_obtener_mascota(cursor, cliente_id, mascota_data)
                cursor.execute("INSERT INTO citas_mascotas (cita_id, mascota_id) VALUES (%s, %s)", 
                             (cita_id, mascota_id))
            
            # Vincular servicios con la cita
            precio_total = 0
            for servicio_id in servicios_ids:
                cursor.execute("INSERT INTO citas_servicios (cita_id, servicio_id) VALUES (%s, %s)", 
                             (cita_id, servicio_id))
                # AquÃ­ podrÃ­as agregar lÃ³gica para calcular precios por servicio
            
            # Actualizar precio total si es necesario
            if precio_total > 0:
                cursor.execute("UPDATE citas SET precio_total = %s WHERE id = %s", (precio_total, cita_id))
        
        conexion.commit()
    except Exception as e:
        conexion.rollback()
        raise e
    finally:
        conexion.close()
    return cita_id

def verificar_o_insertar_cliente(cursor, nombre, email):
    """Verifica si existe un cliente o lo crea"""
    cursor.execute("SELECT id FROM clientes WHERE email = %s", (email,))
    cliente = cursor.fetchone()
    
    if cliente:
        return cliente[0]
    else:
        cursor.execute("INSERT INTO clientes (nombre, email, fecha_registro) VALUES (%s, %s, NOW())", 
                      (nombre, email))
        return cursor.lastrowid

def insertar_o_obtener_mascota(cursor, cliente_id, mascota_data):
    """Inserta o obtiene una mascota existente"""
    nombre = mascota_data.get('nombre', '')
    especie = mascota_data.get('especie', '')
    raza = mascota_data.get('raza', '')
    edad = mascota_data.get('edad')
    peso = mascota_data.get('peso')
    
    # Buscar mascota existente
    cursor.execute("""
        SELECT id FROM mascotas 
        WHERE cliente_id = %s AND nombre = %s AND especie = %s AND activo = 1
    """, (cliente_id, nombre, especie))
    mascota = cursor.fetchone()
    
    if mascota:
        # Actualizar datos si es necesario
        cursor.execute("""
            UPDATE mascotas 
            SET raza = %s, edad = %s, peso = %s 
            WHERE id = %s
        """, (raza, edad, peso, mascota[0]))
        return mascota[0]
    else:
        # Insertar nueva mascota
        cursor.execute("""
            INSERT INTO mascotas (cliente_id, nombre, especie, raza, edad, peso, fecha_registro) 
            VALUES (%s, %s, %s, %s, %s, %s, NOW())
        """, (cliente_id, nombre, especie, raza, edad, peso))
        return cursor.lastrowid

def obtener_cita_por_id(id):
    conexion = obtener_conexion()
    cita = None
    with conexion.cursor() as cursor:
        cursor.execute("""
            SELECT c.id, c.cliente_nombre, c.cliente_email, c.fecha, c.hora, c.estado,
                   c.observaciones, c.precio_total,
                   GROUP_CONCAT(DISTINCT CONCAT(m.nombre, '|', m.especie, '|', COALESCE(m.raza, ''), '|', 
                                              COALESCE(m.edad, ''), '|', COALESCE(m.peso, '')) SEPARATOR ';') as mascotas,
                   GROUP_CONCAT(DISTINCT CONCAT(s.id, '|', s.nombre) SEPARATOR ';') as servicios
            FROM citas c
            LEFT JOIN citas_mascotas cm ON c.id = cm.cita_id
            LEFT JOIN mascotas m ON cm.mascota_id = m.id
            LEFT JOIN citas_servicios cs ON c.id = cs.cita_id
            LEFT JOIN servicios s ON cs.servicio_id = s.id
            WHERE c.id = %s
            GROUP BY c.id
        """, (id,))
        cita = cursor.fetchone()
    conexion.close()
    return cita

def obtener_cita_completa_por_id(cita_id):
    """Obtiene todos los datos de una cita incluyendo mascotas y servicios"""
    conexion = obtener_conexion()
    try:
        with conexion.cursor() as cursor:
            # Obtener datos principales de la cita
            cursor.execute("""
                SELECT id, cliente_nombre, cliente_email, fecha, hora, estado, 
                       precio_total, observaciones
                FROM citas WHERE id = %s
            """, (cita_id,))
            cita_data = cursor.fetchone()
            
            if not cita_data:
                return None
            
            # Convertir a diccionario
            cita = {
                'id': cita_data[0],
                'cliente_nombre': cita_data[1],
                'cliente_email': cita_data[2],
                'fecha': str(cita_data[3]),
                'hora': str(cita_data[4]),
                'estado': cita_data[5],
                'precio_total': float(cita_data[6]) if cita_data[6] else None,
                'observaciones': cita_data[7]
            }
            
            # Obtener mascotas asociadas
            cursor.execute("""
                SELECT m.id, m.nombre, m.especie, m.raza, m.edad, m.peso
                FROM mascotas m
                INNER JOIN citas_mascotas cm ON m.id = cm.mascota_id
                WHERE cm.cita_id = %s
            """, (cita_id,))
            mascotas_data = cursor.fetchall()
            
            cita['mascotas'] = []
            for mascota in mascotas_data:
                cita['mascotas'].append({
                    'id': mascota[0],
                    'nombre': mascota[1],
                    'especie': mascota[2],
                    'raza': mascota[3],
                    'edad': mascota[4],
                    'peso': float(mascota[5]) if mascota[5] else None
                })
            
            # Obtener servicios asociados
            cursor.execute("""
                SELECT s.id, s.nombre, s.descripcion
                FROM servicios s
                INNER JOIN citas_servicios cs ON s.id = cs.servicio_id
                WHERE cs.cita_id = %s
            """, (cita_id,))
            servicios_data = cursor.fetchall()
            
            cita['servicios'] = []
            for servicio in servicios_data:
                cita['servicios'].append({
                    'id': servicio[0],
                    'nombre': servicio[1],
                    'descripcion': servicio[2]
                })
            
            return cita
    
    except Exception as e:
        raise e
    finally:
        conexion.close()

def clientes():
    """Mostrar pÃ¡gina de gestiÃ³n de clientes"""
    if 'usuario_id' not in session or session.get('rol') not in ['admin', 'empleado']:
        flash('No tienes permisos para acceder a esta pÃ¡gina.', 'error')
        return redirect(url_for('dashboard'))
    
    try:
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        # Obtener todos los clientes con sus mascotas
        cursor.execute("""
            SELECT c.id, c.nombre, c.email, c.telefono, c.direccion,
                   c.fecha_registro, c.activo,
                   COUNT(m.id) as total_mascotas,
                   COUNT(ci.id) as total_citas
            FROM clientes c
            LEFT JOIN mascotas m ON c.id = m.cliente_id
            LEFT JOIN citas ci ON c.id = ci.cliente_id
            GROUP BY c.id
            ORDER BY c.nombre
        """)
        clientes = cursor.fetchall()
        
        # EstadÃ­sticas rÃ¡pidas
        cursor.execute("""
            SELECT 
                COUNT(*) as total_clientes,
                COUNT(CASE WHEN activo = TRUE THEN 1 END) as clientes_activos,
                COUNT(CASE WHEN DATE(fecha_registro) = CURDATE() THEN 1 END) as nuevos_hoy
            FROM clientes
        """)
        estadisticas = cursor.fetchone()
        
        cursor.close()
        conexion.close()
        
        return render_template('historial-clientes.html', 
                             clientes=clientes,
                             estadisticas=estadisticas)
    
    except Exception as e:
        flash(f'Error al cargar clientes: {str(e)}', 'error')
        return render_template('historial-clientes.html', clientes=[], estadisticas=(0, 0, 0))

        cursor = conexion.cursor()

            JOIN clientes c ON m.id_cliente = c.id_cliente

                m.raza LIKE %s OR

def obtener_estadisticas_mascotas():

            ORDER BY cantidad DESC
        """)
        por_especie = cursor.fetchall()

        cursor.execute("""

                CASE

            LIMIT 10

def actualizar_cantidad():

    if 'usuario_id' not in session or session.get('rol') not in ['dueÃ±o', 'admin']:
        return jsonify({'error': 'No autorizado'}), 403
    
    try:
        data = request.get_json()
        producto_id = data.get('producto_id')
        tipo_movimiento = data.get('tipo_movimiento')  # 'entrada' o 'salida'
        cantidad = data.get('cantidad')
        motivo = data.get('motivo', '').strip()
        
        if not producto_id or not tipo_movimiento or not cantidad:
            return jsonify({'error': 'Datos incompletos'}), 400
        
        if cantidad <= 0:
            return jsonify({'error': 'La cantidad debe ser mayor a 0'}), 400
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        # Verificar que el producto existe y obtener cantidad actual
        cursor.execute("SELECT nombre, cantidad_actual FROM productos WHERE id = %s", (producto_id,))
        producto = cursor.fetchone()
        if not producto:
            cursor.close()
            conexion.close()
            return jsonify({'error': 'Producto no encontrado'}), 404
        
        nombre_producto, cantidad_actual = producto
        
        # Calcular nuevo cantidad
        if tipo_movimiento == 'entrada':
            nuevo_cantidad = cantidad_actual + cantidad
        elif tipo_movimiento == 'salida':
            if cantidad_actual < cantidad:
                cursor.close()
                conexion.close()
                return jsonify({'error': f'cantidad insuficiente. Disponible: {cantidad_actual}'}), 400
            nuevo_cantidad = cantidad_actual - cantidad
        else:
            cursor.close()
            conexion.close()
            return jsonify({'error': 'Tipo de movimiento invÃ¡lido'}), 400
        
        # Actualizar cantidad
        cursor.execute("""
            UPDATE productos 
            SET cantidad_actual = %s, fecha_actualizacion = NOW()
            WHERE id = %s
        """, (nuevo_cantidad, producto_id))
        
        # Registrar el movimiento (si existe la tabla movimientos_inventario)
        try:
            cursor.execute("""
                INSERT INTO movimientos_inventario 
                (id, tipo_movimiento, cantidad, cantidad_anterior, cantidad_nuevo, 
                 motivo, id_usuario, fecha)
                VALUES (%s, %s, %s, %s, %s, %s, %s, NOW())
            """, (producto_id, tipo_movimiento, cantidad, cantidad_actual, 
                  nuevo_cantidad, motivo, session['usuario_id']))
        except:
            pass  # Si no existe la tabla, continuamos sin registrar el movimiento
        
        conexion.commit()
        cursor.close()
        conexion.close()
        
        return jsonify({
            'success': True,
            'message': f'cantidad actualizado: {nombre_producto} - {cantidad_actual} â†’ {nuevo_cantidad}',
            'cantidad_anterior': cantidad_actual,
            'cantidad_nuevo': nuevo_cantidad
        })
    
    except Exception as e:
        return jsonify({'error': f'Error al actualizar cantidad: {str(e)}'}), 500

def cambiar_estado_servicio():
    """Activar o desactivar un servicio"""
    if 'usuario_id' not in session or session.get('rol') not in ['admin', 'dueÃ±o']:
        return jsonify({'error': 'No autorizado'}), 403
    
    try:
        data = request.get_json()
        servicio_id = data.get('servicio_id')
        activo = data.get('activo')
        
        if servicio_id is None or activo is None:
            return jsonify({'error': 'Datos incompletos'}), 400
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        # Verificar que el servicio existe
        cursor.execute("SELECT nombre FROM servicios WHERE id = %s", (servicio_id,))
        servicio = cursor.fetchone()
        if not servicio:
            cursor.close()
            conexion.close()
            return jsonify({'error': 'Servicio no encontrado'}), 404
        
        # Verificar si hay citas pendientes antes de desactivar
        if not activo:
            cursor.execute("""
                SELECT COUNT(*) FROM citas 
                WHERE id = %s AND estado IN ('pendiente', 'en_proceso')
            """, (servicio_id,))
            citas_pendientes = cursor.fetchone()[0]
            
            if citas_pendientes > 0:
                cursor.close()
                conexion.close()
                return jsonify({
                    'error': f'No se puede desactivar el servicio. Tiene {citas_pendientes} citas pendientes.'
                }), 400
        
        # Cambiar estado
        cursor.execute("""
            UPDATE servicios 
            SET activo = %s, fecha_actualizacion = NOW()
            WHERE id = %s
        """, (activo, servicio_id))
        
        conexion.commit()
        cursor.close()
        conexion.close()
        
        estado_texto = 'activado' if activo else 'desactivado'
        return jsonify({
            'success': True,
            'message': f'Servicio "{servicio[0]}" {estado_texto} correctamente'
        })
    
    except Exception as e:
        return jsonify({'error': f'Error al cambiar estado del servicio: {str(e)}'}), 500

            if not password_actual or not nueva_password or not confirmar_password:
                return jsonify({'error': 'Todos los campos son obligatorios'}), 400
            
            if nueva_password != confirmar_password:
                return jsonify({'error': 'Las contraseÃ±as no coinciden'}), 400
            
            usuario_id = session['usuario_id']
        
        if len(nueva_password) < 4:
            return jsonify({'error': 'La contraseÃ±a debe tener al menos 4 caracteres'}), 400
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        # Verificar contraseÃ±a actual si no es dueÃ±o cambiando otra cuenta
        if session.get('rol') != 'dueÃ±o' or usuario_id == session.get('usuario_id'):
            cursor.execute("SELECT password FROM usuarios WHERE id_usuario = %s", (usuario_id,))
            usuario = cursor.fetchone()
            if not usuario or usuario[0] != password_actual:
                cursor.close()
                conexion.close()
                return jsonify({'error': 'ContraseÃ±a actual incorrecta'}), 400
        
        # Actualizar contraseÃ±a
        cursor.execute("""
            UPDATE usuarios 
            SET password = %s
            WHERE id_usuario = %s
        """, (nueva_password, usuario_id))
        
        conexion.commit()
        cursor.close()
        conexion.close()
        
        return jsonify({
            'success': True,
            'message': 'ContraseÃ±a actualizada correctamente'
        })
    
    except Exception as e:
        return jsonify({'error': f'Error al cambiar contraseÃ±a: {str(e)}'}), 500

def eliminar_usuario():
    """Eliminar un usuario (solo dueÃ±o)"""
    if 'usuario_id' not in session or session.get('rol') != 'dueÃ±o':
        return jsonify({'error': 'No autorizado'}), 403
    
    try:
        data = request.get_json()
        usuario_id = data.get('usuario_id')
        
        if not usuario_id:
            return jsonify({'error': 'ID de usuario requerido'}), 400
        
        # No permitir eliminar al propio dueÃ±o
        if usuario_id == session.get('usuario_id'):
            return jsonify({'error': 'No puedes eliminar tu propia cuenta'}), 400
        
        conexion = obtener_conexion()
        cursor = conexion.cursor()
        
        # Verificar que el usuario existe
        cursor.execute("SELECT username, rol FROM usuarios WHERE id_usuario = %s", (usuario_id,))
        usuario = cursor.fetchone()
        if not usuario:
            cursor.close()
            conexion.close()
            return jsonify({'error': 'Usuario no encontrado'}), 404
        
        username, rol = usuario
        
        # Verificar si tiene registros relacionados
        cursor.execute("SELECT COUNT(*) FROM asistencia WHERE id_usuario = %s", (usuario_id,))
        tiene_asistencias = cursor.fetchone()[0] > 0
        
        if tiene_asistencias:
            cursor.close()
            conexion.close()
            return jsonify({
                'error': 'No se puede eliminar el usuario porque tiene registros de asistencia asociados. Considere desactivarlo en su lugar.'
            }), 400
        
        # Eliminar registros relacionados primero
        cursor.execute("DELETE FROM personal WHERE id_usuario = %s", (usuario_id,))
        cursor.execute("DELETE FROM usuarios WHERE id_usuario = %s", (usuario_id,))
        
        conexion.commit()
        cursor.close()
        conexion.close()
        
        return jsonify({
            'success': True,
            'message': f'Usuario "{username}" eliminado correctamente'
        })
    
    except Exception as e:
        return jsonify({'error': f'Error al eliminar usuario: {str(e)}'}), 500

def mi_perfil():
    """Mostrar y editar perfil del usuario actual"""
    if 'usuario_id' not in session:
        return redirect(url_for('login'))
    
    if request.method == 'GET':
        try:
            conexion = obtener_conexion()
            cursor = conexion.cursor()
            
            # Obtener datos del usuario actual
            cursor.execute("""
                SELECT u.username, u.email, u.rol, u.fecha_creacion,
                       p.nombre_completo, p.telefono, p.direccion, p.fecha_nacimiento
                FROM usuarios u
                LEFT JOIN personal p ON u.id_usuario = p.id_usuario
                WHERE u.id_usuario = %s
            """, (session['usuario_id'],))
            
            usuario = cursor.fetchone()
            cursor.close()
            conexion.close()
            
            if not usuario:
                flash('Usuario no encontrado.', 'error')
                return redirect(url_for('dashboard'))
            
            return render_template('mi-perfil.html', usuario=usuario)
        
        except Exception as e:
            flash(f'Error al cargar perfil: {str(e)}', 'error')
            return redirect(url_for('dashboard'))
    
    else:  # POST - actualizar perfil
        try:
            email = request.form.get('email', '').strip()
            nombre_completo = request.form.get('nombre_completo', '').strip()
            telefono = request.form.get('telefono', '').strip()
            direccion = request.form.get('direccion', '').strip()
            fecha_nacimiento = request.form.get('fecha_nacimiento')
            
            if not email:
                flash('El email es obligatorio.', 'error')
                return redirect(url_for('mi_perfil'))
            
            conexion = obtener_conexion()
            cursor = conexion.cursor()
            
            # Verificar email Ãºnico
            cursor.execute("SELECT id_usuario FROM usuarios WHERE email = %s AND id_usuario != %s", 
                          (email, session['usuario_id']))
            if cursor.fetchone():
                cursor.close()
                conexion.close()
                flash('Ya existe otro usuario con ese email.', 'error')
                return redirect(url_for('mi_perfil'))
            
            # Actualizar usuario
            cursor.execute("""
                UPDATE usuarios 
                SET email = %s
                WHERE id_usuario = %s
            """, (email, session['usuario_id']))
            
            # Actualizar o insertar datos personales
            cursor.execute("SELECT id_usuario FROM personal WHERE id_usuario = %s", (session['usuario_id'],))
            
            if cursor.fetchone():
                # Actualizar datos existentes
                cursor.execute("""
                    UPDATE personal 
                    SET nombre_completo = %s, telefono = %s, direccion = %s, fecha_nacimiento = %s
                    WHERE id_usuario = %s
                """, (nombre_completo, telefono, direccion, fecha_nacimiento, session['usuario_id']))
            else:
                # Insertar nuevos datos
                cursor.execute("""
                    INSERT INTO personal (id_usuario, nombre_completo, telefono, direccion, fecha_nacimiento)
                    VALUES (%s, %s, %s, %s, %s)
                """, (session['usuario_id'], nombre_completo, telefono, direccion, fecha_nacimiento))
            
            conexion.commit()
            cursor.close()
            conexion.close()
            
            flash('Perfil actualizado correctamente.', 'success')
            return redirect(url_for('mi_perfil'))
        
        except Exception as e:
            flash(f'Error al actualizar perfil: {str(e)}', 'error')
            return redirect(url_for('mi_perfil'))
